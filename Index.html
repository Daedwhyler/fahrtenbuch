        function renderTrips() {
            const container = document.getElementById('trips-container');
            const search = document.getElementById('search-input').value.toLowerCase();
            const filterCat = document.getElementById('filter-category').value;

            let filtered = [...trips].sort((a,b) => new Date(b.date + 'T' + (b.time || '00:00')) - new Date(a.date + 'T' + (a.time || '00:00')));

            if (search) {
                filtered = filtered.filter(t => 
                    (t.start && t.start.toLowerCase().includes(search)) ||
                    (t.end && t.end.toLowerCase().includes(search)) ||
                    (t.notes && t.notes.toLowerCase().includes(search)) ||
                    (t.vehicle && t.vehicle.toLowerCase().includes(search))
                );
            }

            if (filterCat) {
                filtered = filtered.filter(t => t.category === filterCat);
            }

            if (filtered.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-12 bg-white dark:bg-ios-darkCard rounded-2xl p-6 border border-gray-100 dark:border-gray-800">
                        <svg class="w-10 h-10 text-ios-gray mx-auto mb-2 opacity-50" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M9 17v-2m3 2v-4m3 4v-6m2 10H7a2 2 0 01-2-2V5a2 2 0 012-2h5.586a1 1 0 01.707.293l5.414 5.414a1 1 0 01.293.707V19a2 2 0 01-2 2z"/></svg>
                        <span class="text-xs font-semibold text-ios-gray block">Keine Fahrten gefunden</span>
                    </div>`;

            container.innerHTML = filtered.map(t => {
                const badgeColor = t.category === 'Geschäftlich' ? 'bg-ios-blue/10 text-ios-blue' : t.category === 'Arbeitsweg' ? 'bg-ios-green/10 text-ios-green' : 'bg-ios-orange/10 text-ios-orange';
                return `
                    <div class="bg-white dark:bg-ios-darkCard p-3.5 rounded-2xl border border-gray-100 dark:border-gray-800 shadow-xs flex flex-col justify-between w-full max-w-full overflow-hidden">
                        <div class="flex items-center justify-between mb-1.5 w-full">
                            <div class="flex items-center space-x-2 truncate">

                            </div>
                            <span class="text-sm font-extrabold tracking-tight text-gray-900 dark:text-white flex-shrink-0 ml-2">${parseFloat(t.distance || 0).toFixed(1)} km</span>
                        </div>

                        <div class="text-xs font-semibold tracking-tight text-gray-800 dark:text-gray-100 truncate w-full">

                            <span class="text-ios-gray truncate">${escapeHtml(t.vehicle || 'Standard')}</span>
                            <div class="flex items-center space-x-3 flex-shrink-0">
                                <button onclick="openModal('${t.id}')" class="text-ios-blue font-semibold active:opacity-60 flex items-center space-x-1">
                                    <span>Bearbeiten</span>

                                    <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"/></svg>

                trips = trips.filter(t => t.id !== pendingDeleteId);
                saveTrips();
                showToast('Fahrt erfolgreich gelöscht');
            }
            closeActionSheet();
        }

        function calculateDist() {
            const start = parseFloat(document.getElementById('form-km-start').value);
            const end = parseFloat(document.getElementById('form-km-end').value);
            if (!isNaN(start) && !isNaN(end) && end >= start) {
                document.getElementById('form-distance').value = (end - start).toFixed(1);
            }
        }

        function handleFormSubmit(e) {
            e.preventDefault();
            const id = document.getElementById('form-id').value;

            const newTrip = {
                id: id || 'trip_' + Date.now(),
                date: document.getElementById('form-date').value,
                time: document.getElementById('form-time').value,
                start: document.getElementById('form-start').value,
                end: document.getElementById('form-end').value,
                kmStart: parseFloat(document.getElementById('form-km-start').value) || null,
                kmEnd: parseFloat(document.getElementById('form-km-end').value) || null,
                distance: parseFloat(document.getElementById('form-distance').value) || 0,
                category: document.getElementById('form-category').value,
                vehicle: document.getElementById('form-vehicle').value,
                notes: document.getElementById('form-notes').value
            };

            if (id) {
                const idx = trips.findIndex(t => t.id === id);
                if (idx !== -1) trips[idx] = newTrip;
                showToast('Fahrt aktualisiert');
            } else {
                trips.push(newTrip);
                showToast('Fahrt gespeichert');
            }

            saveTrips();
            closeModal();
        }

        function updateStats() {
            const totalKm = trips.reduce((s, t) => s + (parseFloat(t.distance) || 0), 0);
            const totalCount = trips.length;

            const bizKm = trips.filter(t => t.category === 'Geschäftlich').reduce((s, t) => s + (parseFloat(t.distance) || 0), 0);
            const comKm = trips.filter(t => t.category === 'Arbeitsweg').reduce((s, t) => s + (parseFloat(t.distance) || 0), 0);
            const privKm = trips.filter(t => t.category === 'Privat').reduce((s, t) => s + (parseFloat(t.distance) || 0), 0);

            document.getElementById('stat-total-km').textContent = totalKm.toFixed(1) + ' km';
            document.getElementById('stat-total-count').textContent = `${totalCount} Fahrten gesamt`;

            document.getElementById('stat-business-km').textContent = bizKm.toFixed(1) + ' km';
            const bizPercent = totalKm > 0 ? Math.round((bizKm / totalKm) * 100) : 0;
            document.getElementById('stat-business-percent').textContent = `${bizPercent}% anteilig`;

            if (totalKm > 0) {
                document.getElementById('bar-business').style.width = `${(bizKm / totalKm) * 100}%`;
                document.getElementById('bar-commute').style.width = `${(comKm / totalKm) * 100}%`;
                document.getElementById('bar-private').style.width = `${(privKm / totalKm) * 100}%`;
            } else {
                document.getElementById('bar-business').style.width = '0%';
                document.getElementById('bar-commute').style.width = '0%';
                document.getElementById('bar-private').style.width = '0%';
            }

            document.getElementById('stat-biz-val').textContent = bizKm.toFixed(0) + ' km';
            document.getElementById('stat-com-val').textContent = comKm.toFixed(0) + ' km';
            document.getElementById('stat-priv-val').textContent = privKm.toFixed(0) + ' km';

            const avg = totalCount > 0 ? totalKm / totalCount : 0;
            const maxKm = trips.length > 0 ? Math.max(...trips.map(t => parseFloat(t.distance) || 0)) : 0;
            const vehicles = new Set(trips.map(t => t.vehicle).filter(Boolean)).size;

            document.getElementById('stat-avg-km').textContent = avg.toFixed(1) + ' km';
            document.getElementById('stat-max-km').textContent = maxKm.toFixed(1) + ' km';
            document.getElementById('stat-vehicle-count').textContent = vehicles || 1;
        }

        function importCSVData() {
            const raw = document.getElementById('csv-text-input').value.trim();
            if (!raw) {
                showToast('Bitte CSV-Daten eingeben');
                return;
            }

            const lines = raw.split(/\r\n|\n/);
            if (lines.length < 1) return;

            const delim = lines[0].includes(';') ? ';' : ',';
            let addedCount = 0;

            lines.forEach((line, index) => {
                if (!line.trim()) return;
                const cols = line.split(delim).map(c => c.replace(/^"|"$/g, '').trim());

                if (index === 0 && (cols[0].toLowerCase().includes('datum') || cols[0].toLowerCase().includes('date'))) return;

                if (cols.length >= 4) {
                    const parsedDate = parseDateVal(cols[0]);
                    const start = cols[1] || 'Import';
                    const end = cols[2] || 'Import';
                    const distance = parseFloat((cols[3] || '0').replace(',', '.')) || 0;
                    const cat = mapCategory(cols[4] || '');
                    const notes = cols[5] || 'Drivenote Import';

                    trips.push({
                        id: 'dn_' + Math.random().toString(36).substr(2, 9),
                        date: parsedDate,
                        time: '12:00',
                        start: start,
                        end: end,
                        kmStart: null,
                        kmEnd: null,
                        distance: distance,
                        category: cat,
                        vehicle: 'Drivenote',
                        notes: notes
                    });
                    addedCount++;
                }
            });

            saveTrips();
            showToast(`${addedCount} Fahrten importiert!`);
            switchTab('trips');
        }

        function mapCategory(str) {
            const s = str.toLowerCase();
            if (s.includes('privat')) return 'Privat';
            if (s.includes('arbeit') || s.includes('pendel')) return 'Arbeitsweg';
            return 'Geschäftlich';
        }

        function parseDateVal(str) {
            if (!str) return new Date().toISOString().slice(0, 10);
            const match = str.match(/^(\d{2})\.(\d{2})\.(\d{4})/);
            if (match) return `${match[3]}-${match[2]}-${match[1]}`;
            return str.slice(0, 10);
        }

        function exportCSV() {
            if (trips.length === 0) {
                showToast('Keine Fahrten vorhanden');
                return;
            }

            let csv = "Datum;Uhrzeit;Start;Ziel;KM_Start;KM_Ende;Distanz_KM;Kategorie;Fahrzeug;Anlass\n";
            trips.forEach(t => {
                csv += `"${t.date}";"${t.time || ''}";"${t.start}";"${t.end}";"${t.kmStart || ''}";"${t.kmEnd || ''}";"${t.distance}";"${t.category}";"${t.vehicle || ''}";"${t.notes || ''}"\n`;
            });

            const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `Fahrtenbuch_Export_${new Date().toISOString().slice(0,10)}.csv`;
            a.click();
            showToast('Export wurde heruntergeladen');
        }

        function formatDate(d) {
            if (!d) return '';
            const parts = d.split('-');
            if (parts.length === 3) return `${parts[2]}.${parts[1]}.${parts[0]}`;
            return d;
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' })[m]);
        }
